nt36xxx I2C touchscreen controller

Required properties:
 - compatible:     "novatek,NVT-ts"
 - gpios: 	   The reset and interrupt GPIO number
 - reg: 	   I2C slave address of the chip
 - virtualkeys:    the virtual keys for back/home/task
 - TP_MAX_X:       the touchpanel size of width
 - TP_MAX_Y:       the touchpanel size of heigth

Example nodes:

&i2c3 {
	status = "okay";

	touchscreen@01{
		compatible = "novatek,NVT-ts";
		reg = <0x01>;
		novatek,reset-gpio = <&ap_gpio 145 GPIO_ACTIVE_HIGH>;
		novatek,irq-gpio = <&ap_gpio 144 GPIO_ACTIVE_HIGH>;
		virtualkeys = <80 1650 30 30
				240 1650 30 30
				400 1650 30 30>;
		TP_MAX_X = <720>;
		TP_MAX_Y = <1600>;
		status = "okay";
	};
};
